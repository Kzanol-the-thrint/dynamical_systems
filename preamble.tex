% Volume and Set Counters
\usepackage{etoolbox}

\newcounter{volume}
\renewcommand\thevolume{\Roman{volume}}
\newcommand{\volume}[1]{%
	\clearpage
	\refstepcounter{volume}%
	\thispagestyle{plain}%
	\begingroup
	\centering \vspace*{3cm}%
	{\Huge\fontsize{25}{30}\selectfont\bfseries Volume~\thevolume\par}%
	\vskip2.5em
	{\Huge\fontsize{20}{25}\selectfont\bfseries #1\par}%
	\vskip4em
	\endgroup
	\addcontentsline{toc}{chapter}{\protect\fontsize{12}{16}\selectfont\protect\bfseries Volume~\thevolume: #1}
	\clearpage
}

\newcounter{set}
\renewcommand\theset{\Roman{set}}
\newcommand{\set}[1]{%
	\clearpage
	\refstepcounter{set}%
	\thispagestyle{plain}%
	\begingroup
	\centering \vspace*{3cm}%
	{\Huge\fontsize{30}{42}\selectfont\bfseries Set~\theset\par}%
	\vskip2.5em
	{\Huge\fontsize{30}{42}\selectfont\bfseries #1\par}%
	\vskip4em
	\endgroup
	\addcontentsline{toc}{chapter}{\protect\fontsize{22}{28}\selectfont\protect\bfseries Set~\theset: #1}%
	\clearpage
}
